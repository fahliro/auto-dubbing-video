# ---------------------------------------------------------------------------
# Pipeline ini sekarang LOKAL untuk STT dan TTS. Satu-satunya panggilan cloud
# adalah LLM penerjemah (lihat LLM_API_KEY / LLM_BASE_URL).
#
# Kenapa tidak pakai layanan TTS/STT cloud:
#   - Edge TTS        : 403. Microsoft memblokir IP datacenter runner GitHub
#                       Actions. Di luar kendali repo, re-run tidak menolong.
#   - Groq Orpheus TTS: tier gratis RPM=10, RPD=100. Video 92 segmen
#                       menghabiskan kuota harian dalam satu run.
#   - Groq Whisper    : RPM=20, RPD=2000, plus harus upload audio tiap run.
#   ctranslate2 (Whisper) dan onnxruntime (Piper) jalan di CPU runner.
#   Runner GitHub yang dipakai repo ini terukur 2 vCPU / 7 GB RAM (Python 3.12),
#   jadi pilih model dengan kepala dingin -- lihat catatan STT_MODEL di workflow.
#   Tanpa API key, tanpa kuota, tanpa rate limit, tak bisa diblokir per IP.
# ---------------------------------------------------------------------------

# STT lokal. faster-whisper sendiri pure-python; yang biner ctranslate2 & av.
# Model di-download sekali lalu di-cache oleh step "Cache model Whisper" di workflow.
faster-whisper>=1.2.0

# WAJIB di-pin < 19. JANGAN di-upgrade tanpa menjalankan workflow lebih dulu.
#
# faster_whisper.audio.decode_audio() memanggil:
#     av.open(input_file, mode="r", metadata_errors="ignore")
# PyAV 19 MENGHAPUS parameter `metadata_errors`, sehingga panggilannya jadi:
#     TypeError: open() got an unexpected keyword argument 'metadata_errors'
# faster-whisper 1.2.1 (rilis terbaru saat ini) belum menyesuaikan.
#
# Ini nyata terjadi di run 38338510340 (Python 3.12 -> av 19.0.1) dan TIDAK
# terdeteksi saat pengujian lokal, karena av 19 tidak menyediakan wheel cp311
# sehingga lingkungan uji Python 3.11 tetap memakai av 18.1.0 dan lolos.
av>=12,<19

# TTS lokal. Wheel cp39-abi3 manylinux x86_64; dependency hanya onnxruntime +
# pathvalidate (TANPA torch). espeak-ng-data sudah dibundel di wheel Linux.
piper-tts>=1.8.0

# LLM penerjemah (OpenAI-compatible). openai 3.x sudah terbukti jalan di run
# yang sukses; batas <4 supaya tidak ada breaking change mayor yang lolos diam-diam.
openai>=1.40.0,<4

# Datang transitif lewat huggingface-hub, tapi dipin eksplisit demi reproducibility.
requests>=2.31.0,<3
