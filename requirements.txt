# ---------------------------------------------------------------------------
# Pipeline ini sekarang LOKAL untuk STT dan TTS. Satu-satunya panggilan cloud
# adalah LLM penerjemah (lihat LLM_API_KEY / LLM_BASE_URL).
#
# Kenapa tidak pakai layanan TTS/STT cloud:
#   - Edge TTS        : 403. Microsoft memblokir IP datacenter runner GitHub
#                       Actions. Di luar kendali repo, re-run tidak menolong.
#   - Groq Orpheus TTS: tier gratis RPM=10, RPD=100. Video 92 segmen
#                       menghabiskan kuota harian dalam satu run.
#   - Groq Whisper    : RPM=20, RPD=2000, plus harus upload audio tiap run.
#   ctranslate2 (Whisper) dan onnxruntime (Piper) jalan di CPU runner 4 vCPU /
#   16 GB: tanpa API key, tanpa kuota, tanpa rate limit, tak bisa diblok per IP.
# ---------------------------------------------------------------------------

# STT lokal. Wheel cp39-abi3 manylinux; dependency ctranslate2 + av + huggingface-hub.
# Model di-download sekali lalu di-cache oleh step "Cache model Whisper" di workflow.
faster-whisper>=1.2.0

# TTS lokal. Wheel cp39-abi3 manylinux x86_64; dependency hanya onnxruntime +
# pathvalidate (TANPA torch). espeak-ng-data sudah dibundel di wheel Linux.
piper-tts>=1.8.0

# LLM penerjemah (OpenAI-compatible). openai 3.x sudah terbukti jalan di run
# yang sukses; batas <4 supaya tidak ada breaking change mayor yang lolos diam-diam.
openai>=1.40.0,<4

# Datang transitif lewat huggingface-hub, tapi dipin eksplisit demi reproducibility.
requests>=2.31.0,<3
